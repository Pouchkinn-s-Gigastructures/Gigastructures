#Triggers situation after game start
country_event = {
    id = giga_scorched.001
    hide_window = yes
    is_triggered_only = yes
    trigger = { has_origin = origin_scorched }

    immediate = {
        #Planet setup
        #Bit of a mess trying to interact in the right orders with the vanilla planet setup parts that apply, could probably be improved
        capital_scope = {
            create_katzenland_deposits_new = yes

            remove_all_buildings = yes
            generate_start_buildings_and_districts = yes
            add_building = building_factory_1 #AI can't survive without, and it's just an annoying rush for players to build a second

            generate_home_planet_final_pass = yes

            add_modifier = {
                modifier = "irradiated_desert_planet"
                days = -1
            }
            if = {
                limit = {
                    root = { is_ai = no }
                }
                add_modifier = {
                    modifier = "irradiated_desert_shelters"
                    years = 15 #long enough to complete the situation and rejig your economy to prevent collapse #TODO make this decay gradually over a long period
                }
            }
            else = {
                add_modifier = {
                    modifier = "irradiated_desert_shelters"
                    years = -1 #AI need more help because of poor planning #TODO when it's changed to decay gradually, this will be unnecessary
                    clear_on_owner_change = yes
                }
            }

            set_planet_flag = giga_scorched_homeworld@root
        }

        #Start story situation
        country_event = { id = giga_scorched.002 days = 720 random = 360 }
        #situation (~12 years) should therefore be over within 15 years of game start
    }
}

#Roll a new random event from the list every x months into the situation
situation_event = {
    id = giga_scorched.003
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        change_variable = {
            which = scorched_situation_timer
            value = 1
        }

        if = {
            limit = {
                check_variable = {
                    which = scorched_situation_timer
                    value >= 36 #roughly 3 years
                }
            }
            export_trigger_value_to_variable = {
                trigger = situation_progress
                variable = scorched_situation_final_position
            }
            owner = {
                set_variable = {
                    which = scorched_situation_final_position
                    value = prev.scorched_situation_final_position
                }
                random_list = {
                    1 = {
                        country_event = { id = giga_scorched.004 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_1 #checks triggered instead of concluded in case it rolls again in the time someone doesn't answer
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.005 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_2
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.006 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_3
                        }
                    }
                    1 = {
                        country_event = { id = giga_scorched.007 }
                        modifier = {
                            factor = 0
                            has_country_flag = triggered_scorched_event_4
                        }
                    }
                }
            }
            set_variable = {
                which = scorched_situation_timer
                value = 0
            }
        }
    }
}

country_event = {
    id = giga_scorched.008
    is_triggered_only = yes

    picture = GFX_evt_unity_symbol
    title = giga_scorched.008.name
    desc = giga_scorched.008.desc

    immediate = {
        set_country_flag = tech_katzen_atomic_munitions_from_scorched #needs to run beforehand to trigger tech swap in option tooltips
    }

    option = {
        trigger = {
            check_variable = {
                which = scorched_situation_final_position
                value < 100
            }
        }
        name = giga_scorched.008.a.renewal
        hidden_effect = {
            remove_country_flag = tech_katzen_atomic_munitions_from_scorched #remove the tech swap if the option with the tech isn't chosen
        }
        add_modifier = {
            modifier = giga_scorched_interstellar_tradition
            days = -1
        }
        capital_scope = {
            remove_modifier = irradiated_desert_planet
            add_modifier = {
                modifier = irradiated_desert_planet_rehabilitated
                days = -1
            }
        }
        give_next_terraforming_tech_option_effect = yes
        every_situation = {
            limit = { is_situation_type = giga_situation_scorched }
            destroy_situation = this
        }
    }

    option = {
        trigger = {
            NOR = { #this is also a backup option
                check_variable = {
                    which = scorched_situation_final_position
                    value < 100
                }
                check_variable = {
                    which = scorched_situation_final_position
                    value > 200
                }
            }
        }
        name = giga_scorched.008.a.synthesis
        add_modifier = {
            modifier = giga_scorched_alloyed_identity
            days = -1
        }
        add_monthly_resource_mult = {
            resource = society_research
            value = @tier4researchreward
            min = @tier4researchmin
            max = @tier4researchmax
        }
        hidden_effect = {
            remove_country_flag = tech_katzen_atomic_munitions_from_scorched #remove the tech swap if the option with the tech isn't chosen
        }
        every_situation = {
            limit = { is_situation_type = giga_situation_scorched }
            destroy_situation = this
        }
    }

    option = {
        trigger = {
            check_variable = {
                which = scorched_situation_final_position
                value > 200
            }
        }
        name = giga_scorched.008.a.continuity
        if = {
            limit = {
                NOT = {
                    has_technology = tech_katzen_atomic_munitions #who is going to have this by this point lmao
                }
            }
            set_country_flag = katzen_atomic_munitions_possible
            add_research_option = tech_katzen_atomic_munitions
            add_tech_progress = {
                tech = tech_katzen_atomic_munitions
                progress = 0.5 #otherwise it's too costly given how early this event occurs
            }
        }
        hidden_effect = {
            set_country_flag = giga_scorched_habitability_preference #opens the Irradiated Desert World Preference trait for modification
        }
        every_owned_planet = {
            limit = {
                has_planet_flag = giga_scorched_homeworld@root
            }
            every_owned_pop_group = {
                modify_species = {
                    species = this
                    add_traits_at_start_of_list = yes
                    ideal_planet_class = pc_katzenland
                    add_trait = trait_pc_katzenland_preference
                }
            }
            hidden_effect = {
                every_owned_pop_group = {
                    modify_species = {
                        species = this
                        remove_trait = trait_pc_desert_preference #very compat much cleanliness
                        remove_trait = trait_pc_arid_preference
                        remove_trait = trait_pc_savannah_preference
                        remove_trait = trait_pc_tropical_preference
                        remove_trait = trait_pc_continental_preference
                        remove_trait = trait_pc_ocean_preference
                        remove_trait = trait_pc_arctic_preference
                        remove_trait = trait_pc_tundra_preference
                        remove_trait = trait_pc_alpine_preference
                        remove_trait = trait_pc_volcanic_preference
                    }
                }
            }
        }
        custom_tooltip = giga_scorched.008.a.continuity.tooltip
        every_situation = {
            limit = { is_situation_type = giga_situation_scorched }
            destroy_situation = this
        }
    }
