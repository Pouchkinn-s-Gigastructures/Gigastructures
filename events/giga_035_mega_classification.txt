namespace = giga_mega_classification

# scopeless yearly sweep to catch unclassified megas
event = {
    id = giga_mega_classification.100
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        every_megastructure = {
            limit = {
                not = {
                    has_megastructure_flag = @giga_mega_classified
                }
            }

            giga_classify_mega_switch = yes
            set_megastructure_flag = @giga_mega_classified
        }
    }
}

# classify mega on construction
country_event = {
    id = giga_mega_classification.200
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        from = {
            giga_classify_mega_switch = yes
            set_megastructure_flag = @giga_mega_classified
        }
    }
}

# re-classify mega on upgrade
country_event = {
    id = giga_mega_classification.300
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        from = {
            if = {
                limit = {
                    has_megastructure_flag = @giga_mega_classified
                }
                giga_classify_mega_clear = yes
            }

            giga_classify_mega_switch = yes
            set_megastructure_flag = @giga_mega_classified
        }
    }
}

# classify constructors on mega build start
country_event = {
    id = giga_mega_classification.400
    is_triggered_only = yes
    hide_window = no #yes

    immediate = {
        fromfrom = {
            if = {
                limit = {
                    has_fleet_flag = @giga_mega_classified
                }
                giga_classify_mega_construction_clear = yes
            }

            giga_classify_mega_construction_switch = yes
            set_fleet_flag = @giga_mega_classified
        }
    }
}
