event = {
	id = giga_blokkat.1500
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOR = { 
			has_global_flag = blokkats_failed_to_spawn
			has_global_flag = blokkats_managed_to_spawn 
			has_global_flag = giga_blokkats_disabled
		}
	}

	immediate = {
		#Check if we can begin the spawning protocols
		event_target:global_event_country = {
			if = {
				limit = { 
					NOR = { 
						has_global_flag = blokkat_trying_to_spawn
						has_global_flag = giga_blokkats_disabled
					} 
				}
				if = {
					limit = { #Default
						check_variable = { which = blokkat_spawn_date value = 2500 }
						OR = {
							end_game_years_passed >= 100
							AND = { end_game_years_passed >= 1 giga_blokkat_defeated_crisis = yes }
						}
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2450
						check_variable = { which = blokkat_spawn_date value = 2450 }
						end_game_years_passed >= 50
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2400
						check_variable = { which = blokkat_spawn_date value = 2400 }
						end_game_years_passed >= 1
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2350
						check_variable = { which = blokkat_spawn_date value = 2350 }
						mid_game_years_passed >= 50
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2300
						check_variable = { which = blokkat_spawn_date value = 2300 }
						mid_game_years_passed >= 1
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2250
						check_variable = { which = blokkat_spawn_date value = 2250 }
						years_passed >= 50
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = {
					limit = { #2250
						check_variable = { which = blokkat_spawn_date value = 2200 }
						years_passed >= 0
					}
					begin_blokkat_spawn_cycle = yes
				}
				else_if = { #Random
					limit = {
						check_variable = { which = blokkat_spawn_date value = 2550 }
						years_passed >= 50
						end_game_years_passed < 100
					}
					random_list = {
						1 = {
							set_global_flag = blokkats_managed_to_spawn
							country_event = { id = giga_blokkat.1000 days = 10 random = 10 }
						}
						99 = {}
					}
				}
				else_if = { #Random but time's up
					limit = {
						check_variable = { which = blokkat_spawn_date value = 2550 }
						end_game_years_passed >= 100
					}
					set_global_flag = blokkats_managed_to_spawn
					country_event = { id = giga_blokkat.1000 days = 10 random = 10 }
				}
			}
			if = { #Run it 10 times, 10% chance each year
				limit = {
					has_global_flag = blokkat_trying_to_spawn
					check_variable = {
						which = blokkat_trying_to_spawn_count
						value < 11
					}
				}
				random_list = {
					7 = {
						set_global_flag = blokkats_managed_to_spawn
						country_event = { id = giga_blokkat.1000 days = 10 random = 10 }
					}
					93 = {}
				}
				change_variable = {
					which = blokkat_trying_to_spawn_count
					value = 1
				}
			}
			if = { #They failed to spawn, sad
				limit = {
					check_variable = {
						which = blokkat_trying_to_spawn_count
						value >= 11
					}
				}
				set_global_flag = blokkats_failed_to_spawn
			}
		}
	}
}

country_event = {
	id = giga_blokkat.1010
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_system = {
			## Make sure they spawn on rim systems AND not ineligible system
			limit = { is_rim_system = yes NOT = { systems_blokkats_wont_stripmine = yes } }
			save_event_target_as = blokkat_spawn_candidate
			set_variable = { which = blokkat_spawn_nearest_player_capital value = 100000 } 	# start high, eventually it'll get clamped down

			every_country = {
				## We don't count nomadic empire into the equation as they can just move away
				limit = { is_ai = no is_nomadic = no }
				capital_scope.solar_system = {
					save_event_target_as = blokkat_player_capital_system
					event_target:blokkat_spawn_candidate = {
						set_variable = { which = blokkat_temp_distance value = value:blokkat_distance_to_target_value|TARGET|blokkat_player_capital_system| }
						## this basically loops until the smallest value possible
						# this value is the distance between the spawn system and the CLOSEST player capital to it, in euclidean
						if = {
							limit = { check_variable = { which = blokkat_temp_distance value < blokkat_spawn_nearest_player_capital } }
							set_variable = { which = blokkat_spawn_nearest_player_capital value = blokkat_temp_distance }
						}
						clear_variable = blokkat_temp_distance
					}
				}
			}
		}

		ordered_system = {
			limit = { is_rim_system = yes NOT = { systems_blokkats_wont_stripmine = yes } }
			position = 0
			order_by = blokkat_spawn_nearest_player_capital
			save_global_event_target_as = blokkat_spawn_system
			star = { save_global_event_target_as = blokkat_spawn_star }
		}

	}
}

						remove_hyperlane_safe = { from = prev to = this }
					}
				}

				clear_orders = yes
				clear_fleet_actions = this

				# An attempt to make the blokkats choose harvesting targets in a fairer way
				# always choosing closest viable system has the unfortunate side-effects of both being random,
				# and beelining the core because the stars are denser there, meaning the distances shrink towards the core
				# - TTFTCUTS, 2022-10-24

				# attempt to intelligently pick a system to move to
				# hopefully this isn't really slow

				# Audience: this will now account for all player capital rather than just one
				# Avoid target that are closer to the player that's closest to the vester

				## Vester's own distance to the nearest player capital system
				event_target:current_blokkat_mothership_system = {
					set_variable = { which = blokkat_vester_to_nearest_player_capital value = 100000 } # start high, eventually it'll get clamped down

						limit = { is_ai = no is_nomadic = no }
						capital_scope.solar_system = {
							save_event_target_as = blokkat_player_capital_system
							prev = {
								set_variable = { which = blokkat_temp_distance value = value:blokkat_distance_to_target_value|TARGET|blokkat_player_capital_system| }
								## this basically loops until the smallest value possible
								# this value is the distance between the vester and the CLOSEST player capital to it, in euclidean
								if = {
									limit = { check_variable = { which = blokkat_temp_distance value < blokkat_vester_to_nearest_player_capital } }
									set_variable = { which = blokkat_vester_to_nearest_player_capital value = blokkat_temp_distance }
								}
								clear_variable = blokkat_temp_distance
							}
						}
					}
				}

				ordered_system = {
					position = 0
					limit = { NOT = { blokkat_invalid_stripmine_target = yes } }
					# distance metric based on several factors including actual distance, distance to every player, core-rim position
					order_by = value:blokkat_system_score
					# smallest value is best
					inverse = yes

					save_event_target_as = blokkat_found_system
				}

				if = {
					# if a target was found, go to that
					limit = { exists = event_target:blokkat_found_system }

					event_target:blokkat_found_system = {
						set_star_flag = future_blokkat_mothership_system
						clear_global_event_target = future_blokkat_mothership_system
						save_global_event_target_as = future_blokkat_mothership_system
						if = {
							limit = { NOT = { has_hyperlane_to = event_target:current_blokkat_mothership_system } }
							add_hyperlane_safe = {
								from = event_target:current_blokkat_mothership_system
								to = THIS
							}
						}
					}

					event_target:blokkat_country = {
						every_owned_fleet = {
							limit = {
								OR = {
									has_fleet_flag = blokkat_stripmine_fleet
									has_fleet_flag = blokkat_mothership_fleet
								}
							}
							set_fleet_flag = blokkat_found_destination
						}
						random_owned_fleet = {
							limit = {
								has_fleet_flag = blokkat_mothership_fleet
							}
							set_timed_fleet_flag = {
								flag = blokkat_mothership_fleet_goes_to_star
								days = 30
							}
						}
					}

					queue_actions = {
						move_to = event_target:blokkat_found_system
					}
				}
				else = {

					# just going to nearest system as fallback
					queue_actions = {
						find_closest_system = {
							trigger = {
								id = giga_blokkat.7003.trigger.1
								NOT = { blokkat_invalid_stripmine_target = yes }
							}
							found_system = {
								effect = {
									id = giga_blokkat.7003.trigger.2
									this = {
										set_star_flag = future_blokkat_mothership_system
										clear_global_event_target = future_blokkat_mothership_system
										save_global_event_target_as = future_blokkat_mothership_system
										if = {
											limit = {
												NOT = {
													has_hyperlane_to = event_target:current_blokkat_mothership_system
												}
											}
											add_hyperlane_safe = {
												from = event_target:current_blokkat_mothership_system
												to = this
											}
										}
									}
									event_target:blokkat_country = {
										every_owned_fleet = {
											limit = {
												OR = {
													has_fleet_flag = blokkat_stripmine_fleet
													has_fleet_flag = blokkat_mothership_fleet
												}
											}
											set_fleet_flag = blokkat_found_destination
										}
										random_owned_fleet = {
											limit = {
												has_fleet_flag = blokkat_mothership_fleet
											}
											set_timed_fleet_flag = {
												flag = blokkat_mothership_fleet_goes_to_star
												days = 30
											}
										}
									}
									save_event_target_as = blokkat_found_system

								}
								move_to = this
							}
						}
					}
				}
			}
			if = {
				limit = {
					has_fleet_flag = blokkat_found_destination
					solar_system? = {
						NOT = { is_same_value = event_target:future_blokkat_mothership_system }
					}
				}
				queue_actions = {
					find_closest_system = {
						trigger = {
							id = giga_blokkat.7003.trigger.3
							is_same_value = event_target:future_blokkat_mothership_system
						}
						found_system = {
							effect = {
								id = giga_blokkat.7003.trigger.4
								this = {
									if = {
										limit = {
											NOT = { has_hyperlane_to = event_target:current_blokkat_mothership_system }
										}
										add_hyperlane_safe = {
											from = event_target:current_blokkat_mothership_system
											to = this
										}
									}
								}
								save_event_target_as = blokkat_found_system
							}
							move_to = this
						}
					}
				}
			}
		}

		# set a trail of flags so we can draw out the route with a debug command
		if = {
			limit = {
				exists = event_target:blokkat_found_system
				not = {
					event_target:current_blokkat_mothership_system = {
						is_same_value = event_target:blokkat_found_system
					}
				}
			}
			event_target:current_blokkat_mothership_system = { 
				set_star_flag = blokkat_vester_path 
				event_target:blokkat_found_system = { 
					set_star_flag = blokkat_vester_path_destination@prev 
				}
			}
		}

		# set flag
		random_system = { 
			limit = {
				any_fleet_in_system = { has_fleet_flag = blokkat_mothership_fleet }
			}
			save_global_event_target_as = current_blokkat_mothership_system
		}
	}
}

	id = giga_blokkat.10001
	hide_window = yes
	is_triggered_only = yes

	immediate = {

		## Loop if the fleet is still in combat
		if = { limit = { is_in_combat = yes }
			every_controlled_ship = {
				giga_combat_prevent_cheesing = yes
			}
			fleet_event = { id = giga_blokkat.10001 days = 1 }

			set_update_modifiers_batch = end
		}
	}
