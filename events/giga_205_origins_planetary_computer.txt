namespace = giga_planetary_computer_origin

### game start
country_event = {
	id = giga_planetary_computer_origin.001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_planetary_computer
		# is_ai = no
	}

	immediate = {
		capital_scope = {
			# clear normal planet stuff we can't use
			clear_deposits = yes
			remove_all_districts = yes
			# add pop to compensate for removed slums and other pop-generating deposits
			create_pop_group = { species = owner size = 100 }

			# add districts
			# housing
			while = {
				count = 5
				add_district = district_giga_pcc_hab
			}

			while = {
				count = 4
				add_blocker = {
					type = d_pcc_origin_blocker
				}
			}

			if = {
				limit = {
					owner = {
						is_regular_empire = no
					}
				}
				add_zone = {
					district = district_giga_pcc_hab
					zone = zone_industrial
					zone_slot = 2
				}
				add_building = {
					building = building_factory_1
					zone = zone_industrial
				}
			}
			else = {
				add_zone = {
					district = district_giga_pcc_hab
					zone = zone_foundry
					zone_slot = 2
				}
			}

			add_zone = {
				district = district_giga_pcc_hab
				zone = zone_habitat_research_unity
				zone_slot = 1
			}

			solar_system = {
				every_system_planet = {
					if = {
						limit = { has_planet_flag = planetary_computer_home_mining_target }
						create_mining_station = { owner = root }
					}
				}
			}

			# adds and removes buildings
			add_building = building_solar_farm
			add_building = building_giga_pcc_scrap_pile

			#gas deposit for exotic metabolism empires
			generate_home_planet_final_pass = yes

			if = { #Galvanic civic game start compat due to deposits being wiped
				limit = { is_galvanic_empire = yes }
				if = {
					limit = { NOT = { has_deposit = d_galvanic_living_metal } }
					add_deposit = d_galvanic_living_metal
				}
			}
		}
	}
}
