namespace = caramel

#void
country_event = {
    id = caramel.1
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        set_variable = {
            which = galaxy_destroyed_portion
            value = trigger:num_galaxy_systems
        }
        multiply_variable = {
            which = galaxy_destroyed_portion
            value = 1
        }
        while = {
            count = galaxy_destroyed_portion
            random_system = {
                limit = {
                    #has_special_star_flag_trigger = no
                    NOT = { has_star_flag = starting_empty_space }
                }
                giga_remove_system_planets = yes
                set_star_class = sc_empty_space
                set_name = "Empty Void"
                set_asteroid_belt = { radius = 0 }
                every_system_planet = {
                    limit = {
                        is_star = yes
                        is_primary_star = no
                    }
                    remove_planet = yes
                }
                star = {
                    if = {
                        limit = { NOT = { has_planet_flag = shroud_storm_changed } }
                        set_planet_flag = shroud_storm_changed
                    }
                    clear_deposits = yes
                    clear_planet_modifiers = yes
                }
                set_star_flag = starting_empty_space
            }
        }
    }
}
