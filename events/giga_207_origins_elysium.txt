namespace = giga_elysium_origin

### Game Start
country_event = {
	id = giga_elysium_origin.001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_elysium
	}

	immediate = {
		# Gives starting technologies
		give_technology = { tech = giga_tech_orbital_elysium message = no }

		# spawn elysium
		capital_scope = {
			set_timed_carrier_flag = {
				flag = ignore_ai_building_limitations
				days = 1
			}

			remove_all_districts = yes
			validate_planet_buildings_and_districts = yes

			solar_system = {
				# create origin platform
				spawn_megastructure = {
					type = orbital_elysium_origin_platform_0
					planet = prev.planet
					owner = root
					graphical_culture = root
				}
				# immediately upgrade it to fire its effects
				random_system_megastructure = {
					limit = {
						is_megastructure_type = orbital_elysium_origin_platform_0
					}
					set_megastructure_flag = origin_elysium
					upgrade_megastructure_to = orbital_elysium_origin_platform_1
					finish_upgrade = yes
				}
			}
			save_event_target_as = giga_elysium_host

			leader = {
				save_event_target_as = giga_elysium_governor
				unassign_leader = this
			}

			# finish the colony and set capital scope
			random_moon = {
				limit = {
					is_planet_class = pc_giga_elysium
				}
				save_event_target_as = giga_elysium

				set_timed_carrier_flag = {
					flag = ignore_ai_building_limitations
					days = 1
				}

				destroy_colony = yes
				giga_get_species_for_colony = {
					TARGET = this
					WHO = root
				}
				create_colony = {
					owner = root
					species = event_target:giga_colony_species
				}
				set_name = {
					key = "GIGA_ELYSIUM_PLANET_NAME"
					variable_string = "[event_target:giga_elysium_host.GetName]"
				}
				add_district = district_giga_elysium_housing
				add_district = district_giga_elysium_control
				if = {
					limit = {
						root = {
							is_low_tech_start = no
						}
					}
					add_district = district_giga_elysium_archive
				}
				set_capital = yes

				# capitals and housing
				if = {
					limit = {
						root = {
							is_regular_empire = yes
						}
					}
					add_building = building_capital
					remove_building = building_colony_shelter
					add_building = building_luxury_residence
					add_building = building_luxury_residence
					add_building = building_luxury_residence

					if = {
						limit = {
							root = {
								is_materialist = yes
							}
						}
						add_zone = {
							district = district_giga_elysium_archive
							zone = zone_research
						}
					}
				}
				else_if = {
					limit = {
						root = {
							is_hive_empire = yes
						}
					}
					add_building = building_hive_warren
				}
				else_if = {
					limit = {
						root = {
							is_machine_empire = yes
						}
					}
					add_building = building_machine_capital
					remove_building = building_deployment_post
					add_building = building_drone_storage
					# event which removes the mysterious extra deployment post which appears on day 2
					carrier_event = {
						id = giga_elysium_origin.003
						days = 2
					}
				}

				# pop growth buildings
				if = {
					limit = {
						root = {
							is_hive_empire = yes
						}
					}
					if = {
						limit = {
							root = {
								has_origin = origin_progenitor_hive
							}
						}
						add_building = building_offspring_nest
					}
					else = {
						add_building = building_spawning_pool
					}
				}
				else_if = {
					limit = {
						root = {
							or = {
								is_individual_machine = yes
								is_machine_empire = yes
							}
						}
					}
					add_building = building_machine_assembly_plant
				}

				# populate the elysium
				create_pop_group = {
					species = event_target:giga_colony_species
					size = @giga_elysium_origin_extra_pops
				}

				check_planet_employment = yes

				prev = {
					giga_resettle_pops = {
						amount = event_target:giga_elysium.trigger:free_jobs
						target = prevprev
						pop_group_predicate = giga_is_owner_main_species
					}
				}

				check_planet_employment = yes
			}

			# adjust homeworld
			while = {
				limit = {
					planet_size < 22
				}
				change_planet_size = 1
			}

			set_update_modifiers_batch = begin

			clear_blockers = yes

			# second slot first for building placement
			if = {
				limit = { root = { giga_has_consumer_goods_civic_or_individualist = yes } }

				add_zone = {
					district = 0
					zone_slot = 2
					zone = zone_industrial
					replace = yes
				}
			}
			else = {
				add_zone = {
					district = 0
					zone_slot = 2
					zone = zone_urban
					replace = yes
				}
				add_building = {
					zone = zone_urban
					building = building_foundry_1
				}
			}

			# first zone second for building placement
			add_zone = {
				district = 0
				zone_slot = 1
				zone = zone_urban
				replace = yes
			}

			# re-fill the other parts, will cause errors because some parts are full but hey
			log_error = "The following errors about failing to add buildings to a planet can be ignored"
			generate_start_buildings_and_districts = yes

			# prune excess city districts, add extra buildings?
			if = {
				limit = { root = { is_regular_empire = yes } }
				giga_prune_district = {
					district = district_city
					at_most = 2
				}
				giga_prune_building = {
					building = building_research_lab_1
					at_most = 0
				}
				add_building = building_luxury_residence

				giga_prune_district = {
					district = district_generator
					at_most = 1
				}
				giga_prune_district = {
					district = district_mining
					at_most = 2
				}
				giga_prune_district = {
					district = district_farming
					at_most = 1
				}
			}
			else_if = {
				limit = { root = { is_hive_empire = yes } }
				giga_prune_district = {
					district = district_hive
					at_most = 1
				}

				giga_prune_district = {
					district = district_generator
					at_most = 1
				}
				giga_prune_district = {
					district = district_mining
					at_most = 2
				}
				giga_prune_district = {
					district = district_farming
					at_most = 1
				}
			}
			else_if = {
				limit = { root = { is_machine_empire = yes } }
				giga_prune_district = {
					district = district_nexus
					at_most = 1
				}

				giga_prune_district = {
					district = district_generator
					at_most = 2
				}
				giga_prune_district = {
					district = district_mining
					at_most = 1
				}
				giga_prune_district = {
					district = district_farming
					at_most = 2
				}
			}

			# move vaults to orbit
			if = {
				limit = { has_building = building_paragon_memory_vaults }
				remove_building = building_paragon_memory_vaults
				event_target:giga_elysium = {
					add_building = building_paragon_memory_vaults
				}
			}

			#gas deposit for exotic metabolism empires
			generate_home_planet_final_pass = yes

			# hopefully force an update
			set_update_modifiers_batch = end
			validate_planet_buildings_and_districts = yes
			check_planet_employment = yes
		}

		assign_governor_effect = yes

		# update stability etc
		country_event = {
			id = giga_elysium.100
		}

		calculate_modifier = yes
	}
}

# colonised planet
carrier_event = {
	id = giga_elysium_origin.002
	hide_window = yes
	is_triggered_only = yes

	# pre_triggers = {
	# 	has_owner = yes
	# }

	trigger = {
		owner = {
			has_origin = origin_elysium
		}
		planet_has_no_megastructure = yes
		nor = { 
			has_carrier_flag = has_orbital_elysium
			giga_is_habitat = yes
			is_non_giga_habitat = yes
		}
	}

	immediate = {
		set_carrier_flag = has_orbital_elysium
		giga_set_has_mega_flag = yes
		solar_system = {
			# create origin platform
			spawn_megastructure = {
				type = orbital_elysium_origin_platform_0
				planet = root
				owner = root.owner
				graphical_culture = root.owner
			}
			# set it upgrading
			random_system_megastructure = {
				limit = {
					is_megastructure_type = orbital_elysium_origin_platform_0
					planet = {
						is_same_value = root
					}
				}
				set_megastructure_flag = origin_elysium
				upgrade_megastructure_to = orbital_elysium_origin_platform_1
			}
		}
	}
}

		}

	immediate = {

	}
}

# planet transfer
planet_event = {
	id = giga_elysium_origin.103
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			AND = {
				FROM = { has_origin = origin_elysium }
				nor = { 
					has_carrier_flag = megastructure
					giga_is_habitat = yes
					is_non_giga_habitat = yes 
					any_moon = {
						is_planet_class = pc_giga_elysium
					}
					has_modifier = giga_origin_no_elysium
				}
			}
			AND = {
				FROM = { NOT = { has_origin = origin_elysium } }
				has_modifier = giga_origin_no_elysium
			}
		}

	}

	immediate = {
		if = {
			limit = {
				FROM = { has_origin = origin_elysium }
			}
			add_modifier = { modifier = giga_origin_no_elysium }
		}
		if = {
			limit = {
				FROM = { NOT = { has_origin = origin_elysium } }
			}
			remove_modifier = giga_origin_no_elysium
		}
	}
}
