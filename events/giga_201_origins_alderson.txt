	id = giga_alderson.003
	is_triggered_only = yes

	picture = GFX_evt_three_sided_deal

	title = giga_alderson.003.name
	desc = giga_alderson.003.desc

	immediate = {
		capital_scope.solar_system = {
			every_system_colony = {
				limit = {
					exists = owner
					owner = {
						is_primitive = yes

							has_country_flag = excognitor_primitives
							has_country_flag = dominion_primitives
						}
					}
					OR = {
						has_planet_flag = excognitor_planet
						has_planet_flag = dominion_planet
					}
				}
				set_owner = root
			}
		}
