# on_tech_increased
# check if nomad, then check if we have the moon or planet net and dont have the event chain
# make it then
# CANT DO IT IN THE TRADEABLE ITEMS BECAUSE FOR SOME REASON IT DOESNT MAKE THE EVENT CHAIN
country_event = {
	id = giga_nomad_events.001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		OR = {
			has_technology = giga_tech_war_moon_2
			has_technology = giga_tech_war_planet
		}
		NOT = { has_event_chain = giga_nomad_permits_evt_chain }
	}

	immediate = {
		begin_event_chain = { event_chain = giga_nomad_permits_evt_chain }
	}
}

# PERMIT REVOKING EVENTS
# note that these also work if the settled empire has a nomad overlord (NOT THE ONE WHICH IS BUILDING IT)

# on_war_beginning
# check if nomad, then check if we have a permit with any enemy, if so revoke
country_event = {
	id = giga_nomad_events.010
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		from = { # the war
			any_war_participant = {
				is_at_war_with = root
				ROOT = {
					has_country_flag = giga_has_war_moon_permit@prev
				}
			}
		}
	}

	immediate = {
		from = {
			every_war_participant = {
				limit = {
					is_nomadic = no
					is_at_war_with = root
					ROOT = {
						has_country_flag = giga_has_war_moon_permit@prev
					}
				}
				# if so remove our permit
				ROOT = {
					giga_remove_war_moon_permit_poi = { SETTLE_ID = @prev }
				}
			}
		}
	}
}

# on_country_destroyed
# check if NOT nomad, then check if any nomad has a permit with us, if so revoke
# doesnt count integration
country_event = {
	id = giga_nomad_events.011
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_country = {
			is_nomadic = yes
			has_country_flag = giga_has_war_moon_permit@root
		}
	}

	immediate = {
		every_country = {
			limit = {
				is_nomadic = yes
				has_country_flag = giga_has_war_moon_permit@root
			}
			giga_remove_war_moon_permit_poi = { SETTLE_ID = @root }
		}
	}
}
