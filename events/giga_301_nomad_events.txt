namespace = giga_nomad_events

###################################################
## nomad permit related
###################################################
# check if nomad, then check if we have a permit with any enemy
# if so revoke
country_event = {
	id = giga_nomad_events.001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		from = {
			any_war_participant = {
				is_nomadic = no
				ROOT = {
					has_country_flag = giga_has_war_moon_permit@prev
				}
			}
		}
	}

	immediate = {
		from = {
			every_war_participant = {
				limit = {
					is_nomadic = no
					ROOT = {
						has_country_flag = giga_has_war_moon_permit@prev
					}
				}
				# if so remove our permit
				ROOT = {
					remove_country_flag = giga_has_war_moon_permit@prev
				}
			}
		}
	}
}
