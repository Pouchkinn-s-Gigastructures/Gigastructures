# on_tech_increased
# check if nomad, then check if we have the moon or planet net and dont have the event chain, make it then
# also add existing variables to the permit
# CANT DO IT IN THE TRADEABLE ITEMS BECAUSE FOR SOME REASON IT DOESNT MAKE THE EVENT CHAIN
country_event = {
	id = giga_nomad_events.001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		OR = {
			has_technology = giga_tech_war_moon_2
			has_technology = giga_tech_war_planet
		}
		NOT = { has_event_chain = giga_nomad_permits_evt_chain }
	}

	immediate = {
		begin_event_chain = { event_chain = giga_nomad_permits_evt_chain }
		every_subject = {
			# every non nomadic subject, give the permanent permit
			limit = {
				is_nomadic = no
			}
			root = {
				giga_add_vassal_celestial_permit_poi = { SETTLE_ID = @prev}
			}
		}
	}
}

# PERMIT REVOKING EVENTS
# note that these also work if the settled empire has a nomad overlord (NOT THE ONE WHICH IS BUILDING IT)
# also handles vassal permit poi revocation

# on_war_beginning
# check if nomad, then check if we have a permit with any enemy, if so revoke
country_event = {
	id = giga_nomad_events.010
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		from = { # the war
			any_war_participant = {
				is_at_war_with = root
				ROOT = {
					has_country_flag = giga_has_war_moon_permit@prev
				}
			}
		}
	}

	immediate = {
		from = {
			every_war_participant = {
				limit = {
					is_at_war_with = root
					ROOT = {
						has_country_flag = giga_has_war_moon_permit@prev
					}
				}
				# if so remove our permit
				ROOT = {
					giga_remove_war_moon_permit_poi = { SETTLE_ID = @prev }
				}
			}
		}
	}
}

# EVENTS SPECIFICALLY FOR VASSAL PERMIT (ALL FOR THE SITUATION LOG, NO GAMEPLAY EFFECT)

# on_becoming_subject
# check if our overlord is a nomad with the event chain, if so give them permanent rights poi
# no country flag needed, just a visual thing
country_event = {
	id = giga_nomad_events.018
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = no
		from = { # overlord
			is_nomadic = yes
			has_event_chain = giga_nomad_permits_evt_chain
		}
	}

	immediate = {
		from = {
			giga_add_vassal_celestial_permit_poi = { SETTLE_ID = @prev}
		}
	}
}

# on_monthly_pulse_country
# scope: nomad
# this is a permit cleanup event for stuff who arent your vassal, but hasnt triggered the other cases
# aka left vassalage but still exists (there isnt an on action)
country_event = {
	id = giga_nomad_events.019
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_nomadic = yes
		has_event_chain = giga_nomad_permits_evt_chain
		any_country = {
			has_any_overlord = no
			root = {
				has_point_of_interest = { poi = giga_celestial_vassal_permit_poi@prev }
			}
		}
	}

	immediate = {
		every_country = {
			limit = {
				has_any_overlord = no
				root = {
					has_point_of_interest = { poi = giga_celestial_vassal_permit_poi@prev }
				}
			}
			root = {
				giga_remove_vassal_celestial_permit_poi = { SETTLE_ID = @prev }
			}
		}
	}
}
