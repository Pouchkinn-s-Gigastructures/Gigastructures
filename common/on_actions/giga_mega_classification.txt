
# catch megas spawned by events which slip past built and upgraded
# no scope
on_yearly_pulse = {
    events = {
        giga_mega_classification.100
    }
}

# classify megas when built
# Root = Country
# From = Megastructure
# FromFrom = System
# FromFromFrom = Fleet
on_megastructure_built = {
    events = {
        giga_mega_classification.200
    }
}

# re-classify megas when upgraded
# Root = Country
# From = Megastructure
# FromFrom = System
on_megastructure_upgraded = {
    events = {
        giga_mega_classification.300
    }
}

# classify constructors when building a mega
# Root = Ship
# From = Country
on_ship_order = {
    events = {
        giga_mega_classification.400
    }
}
