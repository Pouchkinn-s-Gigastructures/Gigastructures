
giga_move_to_capital = {
	optimize_memory
	random_list = {
		4 = { random_planet_within_border = { save_event_target_as = giga_fe_fleet_spawn } }
		2 = { capital_scope.solar_system = { random_system_planet = { save_event_target_as = giga_fe_fleet_spawn } } }
	}
	last_created_fleet = { set_location = event_target:giga_fe_fleet_spawn }
}

giga_create_fe_celestials = {
	optimize_memory
	[[!SYSTEMCRAFT]
	while = {
		count = event_target:global_event_country.giga_$STATE$_$CELESTIAL$craft_min_count
		create_fleet = {
			name = "name_war_$CELESTIAL$"
			settings = { spawn_debris = no }
			effect = {
				set_owner = prev
				create_ship = { fleet_cache_recalc = no name = "giga_$CELESTIAL$_$TYPE$" design = "giga_$CELESTIAL$_$TYPE$" }
			}
		}
		giga_move_to_capital = yes
	}]
	[[HAVE_MAXIMUM]
	while = {
		count = event_target:global_event_country.giga_$STATE$_$CELESTIAL$craft_max_count
		random_list = {
			2 = {
				create_fleet = {
					name = "name_war_$CELESTIAL$"
					settings = { spawn_debris = no }
					effect = {
						set_owner = prev
						create_ship = { fleet_cache_recalc = no name = "giga_$CELESTIAL$_$TYPE$" design = "giga_$CELESTIAL$_$TYPE$" }
					}
				}
			}
			1 = {}
		}
		giga_move_to_capital = yes
	} ]
	[[SYSTEMCRAFT]
	while = {
		count = event_target:global_event_country.giga_$STATE$_systemcraft_count
		create_fleet = {
			name = "name_war_system"
			settings = { spawn_debris = no is_boss = yes }
			effect = {
				set_owner = prev
				create_ship = { fleet_cache_recalc = no name = "giga_system_$TYPE$" design = "giga_system_$TYPE$" }
			}
		}
		giga_move_to_capital = yes
	} ]
}

giga_designate_fe_celestials = {
	optimize_memory
	while = {
		count = event_target:global_event_country.giga_ae_$CELESTIAL$craft_min_count
		if = {
			limit = { any_planet_within_border = { giga_can_be_war_$CELESTIAL$ = yes } }
			random_planet_within_border = {
				limit = { giga_can_be_war_$CELESTIAL$ = yes }
				set_planet_size = $SIZE|10$
				set_planet_flag = dormant_war_$CELESTIAL$
			}
		}
	}
	[[HAVE_MAXIMUM]
	while = {
		count = event_target:global_event_country.giga_ae_$CELESTIAL$craft_max_count
		if = {
			limit = { any_planet_within_border = { giga_can_be_war_$CELESTIAL$ = yes } }
			random_list = {
				2 = {
					random_planet_within_border = {
						limit = { giga_can_be_war_$CELESTIAL$ = yes }
						set_planet_size = $SIZE|10$
						set_planet_flag = dormant_war_$CELESTIAL$
					}
				}
				1 = {}
			}
		}
	} ]
	giga_apply_fe_celestials_buff = yes
}

giga_apply_fe_celestials_buff = {
	optimize_memory
	every_owned_ship = {
		limit = { OR = {
			is_ship_size = giga_war_moon
			is_ship_size = giga_planet_behemoth
			is_ship_size = giga_systemcraft
		} }
		set_ship_flag = giga_moon_fe
		add_modifier = { modifier = giga_fe_planetcraft_buff }
		if = { limit = {
			has_global_flag = giga_fe_planetcrafts_7
			is_ship_size = giga_systemcraft }
			if = { limit = {
				owner = { OR = { is_country_type = fallen_empire is_country_type = awakened_fallen_empire } }
				NOT = { has_modifier = giga_fe_tartarus_systemcraft_buff } }
				add_modifier = { modifier = giga_fe_tartarus_systemcraft_buff }
			}
		}
	}
}

giga_apply_fe_repeatable_modifiers = {
	optimize_memory
	every_country = {
		limit = { is_country_type = fallen_empire }
		if = {
			limit = { NOT = { has_global_flag = gigas_menu_fe_planetcrafts_celestials } }
			change_variable = { which = giga_fe_repeatable_count value = $REPEATABLE_COUNT|0$ }
		}
		## might as well as putting the modifier here
		add_modifier = { modifier = giga_fe_ehof_sr_upkeep_help }
	}
}

# Sets the FE/AE celestial count variable

	optimize_memory
	if = {
		limit = { NOT = { has_global_flag = gigas_menu_fe_planetcrafts_repeatables } }

			set_variable = { which = giga_ae_mooncraft_max_count   value = $AE_MOON_MAX|0$ }

	}
}

# Generic "maybe give this FE a mega" effect
# EVENT: event id duh
# DISABLED_FLAG: global flag that disables the roll
# FACTOR_VALUE and FACTOR_FLAG: optional modifiers that increase or decrease the chance, can add more if needed
# NOTHING_KILL = optional: country flag that zeroes/remove the "nothing" option. TLDR: default is 3 = {} but can be adjusted
giga_fe_roll_generic_mega = {
	optimize_memory
	random_list = {
		1 = {
			country_event = { id = $EVENT$ }
			modifier = { factor = 0 has_global_flag = $DISABLED_FLAG$ }
			modifier = { factor = $FACTOR_VALUE_1|1$ has_country_flag = $FACTOR_FLAG_1|giga_nothing_flag$ }
			modifier = { factor = $FACTOR_VALUE_2|1$ has_country_flag = $FACTOR_FLAG_2|giga_nothing_flag$ }
		}
		3 = {
			modifier = { factor = $NOTHING_KILL|1$ has_country_flag = $NOTHING_KILL_FLAG|giga_nothing_flag$ }
		}
	}
}

giga_hyperlane_obliteration = {
	optimize_memory
	set_spawn_system_batch = begin
	every_system = {
		limit = {
			giga_mass_removal_exclude = yes
		}
		isolate_system = yes
	}
	event_target:pouchkinn_big_brain = {#He done blewed up da lanes
		every_neighbor_system_euclidean = {
			limit = { hyperlane_eligible = yes }
			add_hyperlane_safe = { from = prev to = this }
			every_neighbor_system_euclidean = {
				limit = { hyperlane_eligible = yes }
				add_hyperlane_safe = { from = prev to = this }
			}
		}
	}
	every_system = {
		limit = {
			hyperlane_eligible = yes
		}
		while = {
			count = root.giga_hyperlane_density
			random_neighbor_system_euclidean = {
				limit = { hyperlane_eligible = yes }
				random_list = {
					3 = {}
					1 = { add_hyperlane_safe = { from = prev to = this } }
				}
			}
		} 
	}
	every_system = {
		limit = {
			hyperlane_eligible = yes
			NOT = {
				any_neighbor_system_euclidean = {
					has_hyperlane_to = prev
				}
			}
		}
		while = {
			count = 2
			random_neighbor_system_euclidean = {
				limit = { hyperlane_eligible = yes }
				add_hyperlane_safe = { from = prev to = this }
			}
		}
	}
	set_spawn_system_batch = end
}
