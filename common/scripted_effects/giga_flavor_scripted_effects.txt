create_katzen_socialist = {
	create_pop_group = {
		species = event_target:flusion_katzens
		size = 500
	}
	create_country = {
		species = event_target:flusion_katzens
		name = "Sozialistischer Katzenstaat"
		authority = auth_democratic
		ethos = {
			ethic = ethic_militarist
			ethic = ethic_fanatic_egalitarian
		}
		origin = origin_katzen_civil_war
		civics = {
			civic = civic_shared_burden
			civic = civic_citizen_service
			#later adds civic_crusader_spirit
		}
		flag = {
			icon={
				category="human"
				file="flag_human_15.dds"
			}
			background={
				category="backgrounds"
				file="inverted_v.dds"
			}
			colors={
				"red"
				"burgundy"
				"null"
				"null"
			}
		}
		type = default
		ignore_initial_colony_error = yes
		effect = {
			create_leader = {
				class = official
				species = event_target:flusion_katzens
				name = "Meowkhail Katlinin"
				gender = male
				background_ethic = ethic_militarist
				skill = 5
				traits = {
					trait = katzen_trait_elite
				}
				effect = {
					set_leader_flag = katlinin
					save_global_event_target_as = katzen_sks_ruler
				}
			}
			set_leader = event_target:katzen_sks_ruler
			set_country_flag = katzen_civil_war_socialist
			set_country_flag = katzen_civil_war_faction
			save_global_event_target_as = katzen_civil_war_socialist
			apply_katzen_difficulty = yes
			add_modifier = { #mixed boosts to help transition from abstract to actual economy
				modifier = katzen_splinter_starter_boost
				years = 10
			}
			add_modifier = { #halves ship upkeep until Civil War ends to increase DRAMA (and help them not go broke instantly)
				modifier = katzen_civil_war_navy_boost
				days = -1
			}
			
			every_playable_country = {
				establish_communications_no_message = event_target:katzen_civil_war_socialist
				add_intel = {
					amount = 40
					who = event_target:katzen_civil_war_socialist
				}
			}
			copy_techs_from = {
				target = event_target:flusion_primitives_country
			}
			force_add_civic = civic_crusader_spirit #add third civic now that we have the third civic tech
		}
	}
	set_owner = event_target:katzen_civil_war_socialist
	set_capital = yes
	set_name = "Kittengrad"
	every_system = { #transfer claims
		limit = {
			has_star_flag = katzen_socialist_system
		}
		starbase = {
			set_owner = event_target:katzen_civil_war_socialist
		}
		every_system_planet_colony = {  #claim all other habitable planets in system
			limit = {
				is_capital = no
				owner = {
					is_country_type = katzenartig_imperium
				}
				#NOT = { is_planet_class = pc_giga_siegfried_world } #no point inheriting worlds that will explode
			}
			set_owner = event_target:katzen_civil_war_socialist
		}
		every_system_megastructure = {
			limit = { owner = { is_country_type = katzenartig_imperium } }
			set_owner = event_target:katzen_civil_war_socialist
		}
	}
	event_target:katzen_civil_war_socialist = {
		every_planet_within_border = {
			limit = {
				is_colony = yes
			}
			create_pop_group = {
				species = event_target:flusion_katzens
				size = 200
			}
		}
		add_resource = {
			minerals = 100000
			alloys = 20000
			energy = 100000
			food = 100000
			consumer_goods = 50000
			exotic_gases = 5000
			volatile_motes = 5000
			rare_crystals = 5000
			influence = 250
			unity = 100000 #give a kickstart on traditions since they are lacking
		}
		every_system_within_border = { #claim all katzen fleets in borders
			every_fleet_in_system = {
				limit = {
					owner = {
						is_country_type = katzenartig_imperium
					}
				}
				set_owner = event_target:katzen_civil_war_socialist
			}
		}
	}
}

katzen_reformation_collapse = {
	event_target:flusion_primitives_country = {
		set_update_modifiers_batch = begin
		if = {
			limit = { has_global_flag = giga_storming_pyramid_prep }
			remove_global_flag = giga_storming_pyramid_prep
		}
		if = {
			limit = { has_global_flag = giga_storming_pyramid }
			remove_global_flag = giga_storming_pyramid
		}
		if = {
			limit = { has_global_flag = giga_storming_pyramid_active }
			remove_global_flag = giga_storming_pyramid_active
		}
		set_global_flag = katzenartig_imperium_defeated
		if = {
			limit = { NOT = { any_country = { is_country_type = katzenartig_imperium } } }
			create_country = {
				type = katzenartig_imperium_defeated
				name = "Defeated Katzens"
				effect = { save_global_event_target_as = flusion_primitives_country }
			}
		}
		clear_flusion_ezic_sabotage_modifiers = yes
		clear_flusion_katown_shield_modifiers = yes
		capital_scope = { remove_modifier = flusion_consolidation_instability }
		if = { #all remaining Siegfrieds blow up
			limit = {
				any_galaxy_planet = { is_planet_class = pc_giga_siegfried_world }
			}
			event_target:global_event_country = {
				country_event = { id = giga_katzen.123 days = 15 }
			}
		}
		every_system = { #scuttle all submarines
			every_ship_in_system = {
				limit = {
					OR = {
						is_ship_size = giga_katzen_submarine
						is_ship_size = giga_katzen_submarine_submerged
					}
				}
				destroy_ship = THIS
			}
		}
		if = {
			limit = { exists = event_target:katzen_submarine_country }
			event_target:katzen_submarine_country = { destroy_country = yes }
		}
		every_country = {
			limit = { has_event_chain = "katzenartig_chain" }
			end_event_chain = "katzenartig_chain"
		}
		every_country = {
			limit = { has_modifier = galcom_mobilizes_against_kaiser }
			remove_modifier = galcom_mobilizes_against_kaiser
		}
		if = {
			limit = { has_global_flag = galcom_kaiser_crusade }
			remove_global_flag = galcom_kaiser_crusade
		}
		every_playable_country = {
			limit = { has_modifier = katzen_attache }
			remove_modifier = katzen_attache
		}
		every_country = {
			clear_contribute_ezic_modifiers = yes
		}
		if = {
			limit = {
				exists = event_target:flusion_douclaw
			}
			event_target:flusion_douclaw = {
				unassign_leader = this
			}
		}
		if = {
			limit = { has_global_flag = generating_katzen_wormhole }
			remove_global_flag = generating_katzen_wormhole
		}
		clear_global_event_target = katzen_wormhole_target
		every_megastructure = {
			limit = {
				is_megastructure_type = katzen_ehof_wormhole
			}
			delete_megastructure = THIS
		}
		every_megastructure = { #Delete any in building kaiser moons in system with ONLY hostile ships
			limit = {
				is_megastructure_type = kaiser_moon_1
				owner = { is_country_type = katzenartig_imperium }
			}
			delete_megastructure = THIS
		}
		every_megastructure = {
			limit = {
				OR = {
					is_megastructure_type = katzen_forge_0
					is_megastructure_type = katzen_forge_1
				}
