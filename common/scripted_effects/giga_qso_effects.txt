giga_qso_setup_owner = {
    if = {
        limit = {
            NOT = {
                has_country_flag = giga_qso_no_owner
            }
        }
        random_country = {
            limit = {
                has_country_flag = giga_qso_target_country@root
            }
            save_global_event_target_as = giga_qso_target_country
        }
    }
}

giga_qso_option_internal_if = {
    if = {
        limit = {
            NOT = {
                has_country_flag = giga_qso_no_owner
            }
        }
        event_target:giga_qso_target_country = {
            if = {
                limit = {
                    NOT = {
                        is_country_type = primitive
                    }
                }
                every_system_within_border = {
                    limit = {
                        ndb_can_target_system = yes
                        or = {
                            [[condition]	$condition|always$ = yes ]
                            [[condition2]
                                $condition2$ = yes
                            ]
                            [[condition3]
                                $condition3$ = yes
                            ]
                            [[condition4] $condition4$ = yes  ]
                        }
                    }
                    giga_event_target_list_save = yes
                }
            }
            else = {
                random_owned_planet = {
                    solar_system = {
                        giga_event_target_list_save = yes
                    }
                }
            }
        }
    }
    else = {
        every_system = {
            limit = {
                ndb_can_target_system = yes
                NOT = {
                    exists = owner
                }
                or = {
                    [[condition]	$condition|always$ = yes ]
                    [[condition2]
                        $condition2$ = yes
                    ]
                    [[condition3]
                        $condition3$ = yes
                    ]
                    [[condition4] $condition4$ = yes]
                }
            }
            giga_event_target_list_save = yes
        }
    }
}

    every_system = {
        limit = { has_star_flag = giga_qso_target@root }
        remove_star_flag = giga_qso_target@root
    }
    set_star_flag = giga_qso_target@root
    root = {
        set_country_flag = giga_ndb_firing_system
        country_event = { id = giga_ndb.201 days = 2 }
    }
}
