# Standard Candle system
# A pair of systems specially spawned to allow calculation of the absolute locations of any solar system
# Important for finding angles between systems for bisecting hyperlanes

# spawn a single candle system, set variables and global event targets
giga_spawn_standard_candle = {
    inline_script = {
        script = systems/standard_candles/spawn_standard_candle

        min_distance = 20
        max_distance = 500

        distance_steps = 12 #3
        rotation_steps = 36 #8
    }
}

# spawn both candle systems, at distances appropriate for the galaxy size (or a best guess)
giga_spawn_standard_candles = {
    # general fallback values
    set_variable = { which = giga_candle_min_distance value = 40 }
    set_variable = { which = giga_candle_max_distance value = 300 }

    # set variables from galaxy size and shape to try and avoid weird placements


    # do the spawning
    # the spawn effect handles setting variables and event targets
    set_spawn_system_batch = begin
    giga_spawn_standard_candle = yes

    # refine the distance ranges so the second one is less likely to be out of place


    # spawn second system
    reroll_random = yes
    giga_spawn_standard_candle = yes
    set_spawn_system_batch = end

    # clean up
    clear_variable = giga_candle_min_distance
    clear_variable = giga_candle_max_distance
}
