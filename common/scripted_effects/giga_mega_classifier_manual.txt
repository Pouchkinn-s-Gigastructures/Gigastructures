# default case for the main classifier switch
giga_classify_mega_switch_default = {
    if = {
        limit = {
            third_party_kilostructure_list = { CONDITION = is_megastructure_type }
        }
        set_megastructure_flag = giga_tag_kilostructure
    }
    else_if = {
        limit = {
            third_party_gigastructure_list = { CONDITION = is_megastructure_type }
        }
        set_megastructure_flag = giga_tag_gigastructure
    }
    else = {
        set_megastructure_flag = giga_tag_megastructure
    }
}

# default case for the construction classifier switch
giga_classify_mega_construction_switch_default = {
    if = {
        limit = {
            third_party_kilostructure_list = { CONDITION = is_constructing }
        }
        set_fleet_flag = giga_tag_kilostructure
    }
    else_if = {
        limit = {
            third_party_gigastructure_list = { CONDITION = is_constructing }
        }
        set_fleet_flag = giga_tag_gigastructure
    }
    else = {
        set_fleet_flag = giga_tag_megastructure
    }
}
