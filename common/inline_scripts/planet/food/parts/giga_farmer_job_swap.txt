# regular empires
inline_script = {
    script = generic_parts/giga_toggled_job
    job = farmer
    add = $add$
    type = $type$
    count = $jobs$
    condition = "
        owner = {
            is_gestalt = no
            is_fallen_empire_spiritualist = no
        }
        $condition$
    "
    multiplier_string = "$multiplier_string$"
}
inline_script = {
    script = generic_parts/giga_toggled_desc
    toggle = @[ $jobs$ * $description$ ]
    trigger = "
        owner = {
            is_gestalt = no
            is_fallen_empire_spiritualist = no

            #farmer_is_titan_hunter_trigger = no
            #farmer_is_puddle_technician_trigger = no
            farmer_is_angler_trigger = no
        }
        $condition$
    "
    text = job_farmer_effect_desc
}
