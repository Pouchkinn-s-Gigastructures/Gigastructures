# Gigastructures: contract-free copy of vanilla
# common/inline_scripts/scripted_actions/harvest_resources_from_deposit.txt (Stellaris v4.5.2).
# Use it for resources that have no contract_harvest_<resource> mission type or event chain.
# Calling the vanilla inline for those would reference contract and event chain keys that do not exist.

# Parameters are the same as the vanilla inline:
#	RESOURCE: the resource key
#	BASE: @arkship_harvest_resources_duration (Arkship) or @logistic_harvest_resources_duration (Logistic Ship)
#	TARGET: event_target:arkship.controller (Arkship) or event_target:arkship, the ship itself (Logistic Ship)
#	EFFECT: add_resource (Arkship) or add_resource_to_local_stockpile (Logistic Ship)
# Scope: the harvested planet. Relies on event_target:harvesting_ship, saved by arkship_harvest_resources_effect.

if = {
	limit = {
		has_resource = { type = $RESOURCE$ amount > 0 }
	}
	$TARGET$ = {
		$EFFECT$ = {
			$RESOURCE$ = 1
			mult = root.value:arkship_harvest_resource_from_deposit_value|RESOURCE|$RESOURCE$|BASE|$BASE$|MULT|1|
		}
	}
}
