on_queued = {
    owner? = {
        change_variable = { which = $MEGA_CURRENT_VARIABLE$ value = 1 }
    }
    $ON_QUEUED$
}

on_unqueued = {
    owner? = {
        if = {
            limit = {
                check_variable = { which = $MEGA_CURRENT_VARIABLE$ value >= 1 }
            }
            change_variable = { which = $MEGA_CURRENT_VARIABLE$ value = -1 }
        }
    }
    $ON_UNQUEUED$
}

abort_construction_trigger = {
    owner? = {
        check_variable = {
            which = $MEGA_CURRENT_VARIABLE$
            value >= owner.value:$MEGA_LIMIT_VALUE$
        }
    }
    $ABORT_CONSTRUCTION_TRIGGER$
}

on_aborted = {
    owner? = {
        if = {
            limit = {
                check_variable = { which = $MEGA_CURRENT_VARIABLE$ value >= 1 }
            }
            change_variable = { which = $MEGA_CURRENT_VARIABLE$ value = -1 }
        }
    }
    $ON_ABORTED$
}

on_destroyed = {
    owner? = {
        if = {
            limit = {
                check_variable = { which = $MEGA_CURRENT_VARIABLE$ value >= 1 }
            }
            change_variable = { which = $MEGA_CURRENT_VARIABLE$ value = -1 }
        }
    }
    $ON_DESTROYED$
}
