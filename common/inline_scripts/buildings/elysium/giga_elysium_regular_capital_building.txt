inline_script = {
    script = buildings/elysium/giga_elysium_shared_capital_building
    level = $level$
    level_plus_one = @[ $level$ + 1 ]

    potential = "
        owner? = {
            is_gestalt = no
        }
    "
}

######################################################
# Modifiers

planet_modifier = {
    planet_housing_add = @[ 500 * $level$ ]
    planet_amenities_add = @[ 500 * $level$ ]
    job_bureaucrat_add = @[ 150 * $level$ ]
    job_enforcer_add = @[ 200 * $level$ ]

    planet_max_branch_office_buildings_add = $level$
    planet_defense_armies_add = @[ 4 * $level$ ]
}

######################################################
# General capital jobs

# Synthetics Prefabricated Components
# +100 Roboticist
triggered_planet_modifier = {
    potential = {
        exists = owner
        owner = {
            is_regular_empire = yes
            has_active_tradition = tr_synthetics_prefabricated_components
        }
    }
    job_roboticist_add = 100
}


# Individualist Machines
# +200 Roboticist
triggered_planet_modifier = {
    potential = {
        exists = owner
        owner = {
            is_individual_machine = yes
        }
    }
    job_roboticist_add = 200
}

# Pharma State
# +100 Medical Worker
triggered_planet_modifier = {
    potential = {
        exists = owner
        owner = { has_valid_civic = civic_private_healthcare_corporate }
    }
    job_healthcare_add = 100
}

######################################################
# Politician replacement modifiers

# Cyber Imperial Jobs counterbalance
triggered_planet_modifier = {
    potential = {
        exists = owner
        owner = {
            is_cyber_imperial = yes
        }
    }
    job_politician_add = @[ -1 * building_static_jobs ]
}

# Imperious Architecture counterbalance
triggered_planet_modifier = {
    potential = {
        exists = owner
        owner = {
            is_gestalt = no
            is_worker_coop_empire = no
            has_active_tradition = tr_domination_imperious_architecture
        }
    }
    job_politician_add = -100
    job_bureaucrat_add = 150
    mult = value:imperious_architecture_scaling_value
}
