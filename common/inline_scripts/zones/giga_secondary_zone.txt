# Secondary (arcology, ringworld, nexus, hive) zone for a giga resource zone
# ZONE = zone name suffix, zone_giga_$ZONE$ (also used for the jobs script and swap district)
# ICON = district specialization icon suffix
# SET = building set the zone allows
# RESOURCE = resource the AI expects it to produce
# TECH = tech that unlocks the zone
# TYPE = arcology / ring_world / nexus / hive (zone set and swap district)
# MODIFIERS = arcology / ring_world (housing modifiers to use)

icon = GFX_giga_district_specialization_$ICON$
base_buildtime = @zone_buildtime
planet_limit = 1

potential = { # planet scope
	hidden_trigger = { exists = owner }
	owner = {
		has_technology = $TECH$
	}
}

unlock = { # planet scope
	hidden_trigger = { exists = owner }
	owner = {
		has_technology = $TECH$
	}
}

resources = {
	category = planet_zones
	cost = {
		minerals = @zone_cost
	}
}

inline_script = {
	script = jobs/giga_zone_$ZONE$_add
	AMOUNT = @doubled_scaling_district_1_job
}

inline_script = {
	script = zones/shared_$MODIFIERS$_zone_modifiers
}

swap_type = district_$TYPE$_giga_$ZONE$
swap_type_weight = 5

zone_sets = {
	$TYPE$
}

included_building_sets = {
	$SET$
}

convert_to = {
	zone_giga_$ZONE$
	zone_giga_$ZONE$_arcology
	zone_giga_$ZONE$_ring_world
	zone_giga_$ZONE$_nexus
	zone_giga_$ZONE$_hive
}

planet_modifier = {
	zone_building_slots_add = 3
}

ai_resource_production = {
	$RESOURCE$ = 1
	trigger = { always = yes }
}

ai_priority = 2
