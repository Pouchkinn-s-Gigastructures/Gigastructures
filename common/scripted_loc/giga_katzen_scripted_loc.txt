define_text = {
    name = KatzenAttacksNationTitle
    default = giga_katzen.051.default.name
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = no
                has_federation = yes
            }
        }
        localization_key = giga_katzen.051.federation.name
    }
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = yes
                has_country_flag = giga_meopa_fe
            }
        }
        localization_key = giga_katzen.051.precursor_katzen.name
    }
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = yes
                NOT = { has_country_flag = giga_meopa_fe }
            }
        }
        localization_key = giga_katzen.051.precursor_nokatzen.name
    }
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = yes
                NOT = { has_country_flag = giga_meopa_fe }
            }
        }
        localization_key = giga_katzen.051.precursor_nokatzen.name
    }
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = no #niche circumstances (mostly modded) allow FE subjects but we don't account for that yet
                any_subject = { is_same_value = root }
            }
        }
        localization_key = giga_katzen.051.overlordofplayer.name
    }
    text = {
        trigger = {
            event_target:kaiser_attack_target = {
                is_fallen_empire = no #niche circumstances allow FE overlords but we don't account for that yet
                overlord? = { is_same_value = root }
            }
        }
        localization_key = giga_katzen.051.subjectofplayer.name
    }
}
