swaystation_piracy = {
	potential = {
		is_waystation_starbase = yes
		can_build_pirate_waystation_module = yes
		OR = {
			# AI: target rival systems with high-pop colonies
			AND = {
				owner = {
					is_ai = yes
				}
				fleet = {
					solar_system = {
						owner? = {
							is_rival = root.owner
						}
						any_system_planet = {
							is_colony = yes
							pop_amount >= 2500
						}
					}
				}
			}
			# Player: trigger when pirate modules are already present
			AND = {
				owner = {
					is_ai = no
				}
				OR = {
					has_starbase_module = waystation_pirate_hideout
					has_starbase_module = waystation_concealment_field
					has_starbase_module = waystation_reaver_stronghold
				}
			}
		}
	}
	is_waystation = yes		# Use the Wayline UI for the outliner instead of the Starbase UI.

	weight_modifier = {
		base = @high_prio
	}

	# Piracy waystation: prioritize pirate, defense, and cloaking modules
	inline_script = {
