gigas_trade_action_nomad_war_moon_permit = {
	fire_and_forget = yes

	# the permit flag is stored on the requester (nomad)
	# permit stores the ID of the settled country
	potential = {
		has_nomads_dlc = yes
		# GIVER is settled, ASKER is nomad
		FROM = {
			is_nomadic = yes
			has_technology = giga_tech_war_moon_1
			NOT = { has_country_flag = giga_has_war_moon_permit@prev}
			NOT = { is_overlord_to = root }
		}
		# if GIVER is settled, or any settled vassals (for nomad GIVER with vassals)
		OR = {
			is_nomadic = no
			any_subject = {
				is_nomadic = no
			}
		}
		has_any_overlord = no
		# no trading with FEs or other weird ppl
		is_country_type = default
	}

	on_traded_effect = {
		FROM = {
			giga_add_war_moon_permit_poi = { SETTLE_ID = @root }
		}
	}

	ai_weight = {
		weight = 800
		# harder to ask from a nomad
		modifier = {
			factor = 1.25
			is_nomadic = yes
		}
	}
}
