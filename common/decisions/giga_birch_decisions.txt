# Origin Birch Restoration

decision_giga_origin_birch_restore = {
	owned_planets_only = yes
	icon = decision_birch_world

	enactment_time = 1800
	resources = {
		category = decisions
		cost = {
			alloys = 40000
			unity = 5000
			influence = 250
		}
	}

	potential = {
		is_planet_class = pc_origin_birch
		has_carrier_flag = birch_expedition_4_complete # foundry
		has_carrier_flag = birch_expedition_7_complete # observatory
		has_carrier_flag = birch_expedition_9_complete # extraction
		has_carrier_flag = birch_expedition_10_complete # final push
		owner? = {
			has_origin = origin_birch
		}
	}

	abort_trigger = {
		or = {
			not = { exists = owner }
			owner = {
				not = {
					has_origin = origin_birch
				}
			}
		}
	}

	effect = {
		tooltip = {
			change_pc = pc_birch
		}
		custom_tooltip = decision_giga_origin_birch_restore_tooltip

		hidden_effect = {
			# set flag for speical effects like minimum insulae
			set_carrier_flag = giga_origin_birch

			############################################################################
			# gather information for district replacement

			# city districts
			set_variable = {
				which = birch_districts_city
				value = 0
			}
			if = {
				limit = {
					owner = { is_gestalt = no }
				}
				export_trigger_value_to_variable = {
					trigger = num_districts
					parameters = {
						type = district_city
					}
					variable = birch_districts_city
				}
			}
			else_if = {
				limit = {
					owner = { is_hive_empire = yes }
				}
				export_trigger_value_to_variable = {
					trigger = num_districts
					parameters = {
						type = district_hive
					}
					variable = birch_districts_city
				}
			}
			else_if = {
				limit = {
					owner = { is_machine_empire = yes }
				}
				export_trigger_value_to_variable = {
					trigger = num_districts
					parameters = {
						type = district_nexus
					}
					variable = birch_districts_city
				}
			}

			# city zones
			# so num_zones = (1 if in slot 1) + (1 if in slot 2)
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_foundry
				}
				variable = birch_zones_foundry
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_factory
				}
				variable = birch_zones_factory
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_industrial
				}
				variable = birch_zones_industrial
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_trade
				}
				variable = birch_zones_trade
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_giga_supertensile
				}
				variable = birch_zones_supertensiles
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_unity
				}
				variable = birch_zones_unity
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_unity_spiritualist
				}
				variable = birch_zones_unity_spiritualist
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_research_unity
				}
				variable = birch_zones_research_unity
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_unity_bio_trophy
				}
				variable = birch_zones_sanctuary
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_research
				}
				variable = birch_zones_research
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_research_physics
				}
				variable = birch_zones_research_physics
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_research_society
				}
				variable = birch_zones_research_society
			}
			export_trigger_value_to_variable = {
				trigger = num_zones
				parameters = {
					type = zone_research_engineering
				}
				variable = birch_zones_research_engineering
			}

			# foundry ( num foundry zones + half num industry zones ) * city districts
			set_variable = {
				which = birch_weight_foundry
				value = birch_zones_industrial
			}
			multiply_variable = {
				which = birch_weight_foundry
				value = 0.5
			}
			change_variable = {
				which = birch_weight_foundry
				value = birch_zones_foundry
			}
			multiply_variable = {
				which = birch_weight_foundry
				value = birch_districts_city
			}
			# factory ( num factory zones + half num industry zones ) * city districts
			set_variable = {
				which = birch_weight_factory
				value = birch_zones_industrial
			}
			multiply_variable = {
				which = birch_weight_factory
				value = 0.5
			}
			change_variable = {
				which = birch_weight_factory
				value = birch_zones_factory
			}
			multiply_variable = {
				which = birch_weight_factory
				value = birch_districts_city
			}
			# trade ( num trade zones ) * city districts
			set_variable = {
				which = birch_weight_trade
				value = birch_zones_trade
			}
			multiply_variable = {
				which = birch_weight_trade
				value = birch_districts_city
			}
			# supertensiles ( num supertensile zones ) * city districts
			set_variable = {
				which = birch_weight_supertensiles
				value = birch_zones_supertensiles
			}
			multiply_variable = {
				which = birch_weight_supertensiles
				value = birch_districts_city
			}
			# unity ( num unity zones + num spiritualist unity zones + half num research-unity zones ) * city districts
			set_variable = {
				which = birch_weight_unity
				value = birch_zones_research_unity
			}
			multiply_variable = {
				which = birch_weight_unity
				value = 0.5
			}
			change_variable = {
				which = birch_weight_unity
				value = birch_zones_unity
			}
			change_variable = {
				which = birch_weight_unity
				value = birch_zones_unity_spiritualist
			}
			multiply_variable = {
				which = birch_weight_unity
				value = birch_districts_city
			}
			# sanctuary ( num organic sanctuary zones ) * city districts
			set_variable = {
				which = birch_weight_sanctuary
				value = birch_zones_sanctuary
			}
			multiply_variable = {
				which = birch_weight_sanctuary
				value = birch_districts_city
			}
			# research ( num research zones of any kind + half num research-unity zones ) * city districts
			# 2 * lab districts is added in giga_birch_restoration_physma_laboratory
			set_variable = {
				which = birch_weight_research
				value = birch_zones_research_unity
			}
			multiply_variable = {
				which = birch_weight_research
				value = 0.5
			}
			change_variable = {
				which = birch_weight_research
				value = birch_zones_research
			}
			change_variable = {
				which = birch_weight_research
				value = birch_zones_research_physics

				which = birch_weight_research
				value = birch_zones_research_society
			}

				which = birch_weight_research
				value = birch_zones_research_engineering
			}
			multiply_variable = {
				which = birch_weight_research
				value = birch_districts_city
			}

			export_trigger_value_to_variable = {

				trigger = num_districts

			add_deposit = d_giga_birch_mechanics

				which = giga_birch_district_insula_built
				value = 3
			}

			# orykta
			while = {
				count = value:giga_birch_restoration_orykta_generator

				set_or_change_variable = {
					which = giga_birch_district_orykta_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_orykta_mining
				add_district = district_giga_birch_orykta_minerals
				set_or_change_variable = {
					which = giga_birch_district_orykta_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_orykta_dark_matter
				add_district = district_giga_birch_orykta_dark_matter
				set_or_change_variable = {
					which = giga_birch_district_orykta_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_orykta_negative_mass
				add_district = district_giga_birch_orykta_negative_mass
				set_or_change_variable = {
					which = giga_birch_district_orykta_built
					value = 1
				}
			}
				# no refinery zone to convert so leave it empty

			# prune orykta
			while = {
				limit = {
					check_variable = {
						which = giga_birch_district_orykta_built
						value > 10
					}
				}

				# energy first
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_orykta_built
							value > 10
						}

							type = district_giga_birch_orykta_energy
							value > 0
						}

					set_or_change_variable = {
						which = giga_birch_district_orykta_built
						value = -1
					}

					limit = {
						check_variable = {
							which = giga_birch_district_orykta_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_orykta_minerals
							value > 0
						}
					}
					remove_district = district_giga_birch_orykta_minerals
					set_or_change_variable = {
						which = giga_birch_district_orykta_built
						value = -1
					}
				}

				# no others since the only other possibility should be exactly 1 dark matter sector
			}

			# ktisma
			while = {
				count = value:giga_birch_restoration_ktisma_factory
				add_district = district_giga_birch_ktisma_factory
				set_or_change_variable = {
					which = giga_birch_district_ktisma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_ktisma_foundry
				add_district = district_giga_birch_ktisma_foundry
				set_or_change_variable = {
					which = giga_birch_district_ktisma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_ktisma_trade
				add_district = district_giga_birch_ktisma_trade
				set_or_change_variable = {
					which = giga_birch_district_ktisma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_ktisma_sentient_metal
				add_district = district_giga_birch_ktisma_sentient_metal
				set_or_change_variable = {
					which = giga_birch_district_ktisma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_ktisma_supertensiles
				add_district = district_giga_birch_ktisma_supertensiles
				set_or_change_variable = {
					which = giga_birch_district_ktisma_built
					value = 1
				}
			}

			# prune ktisma
			while = {
				limit = {
					check_variable = {
						which = giga_birch_district_ktisma_built
						value > 10
					}
				}

				# foundry first
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_ktisma_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_ktisma_foundry
							value > 0
						}
					}
					remove_district = district_giga_birch_ktisma_foundry
					set_or_change_variable = {
						which = giga_birch_district_ktisma_built
						value = -1
					}
				}
				# factory second
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_ktisma_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_ktisma_factory
							value > 0
						}
					}
					remove_district = district_giga_birch_ktisma_factory
					set_or_change_variable = {
						which = giga_birch_district_ktisma_built
						value = -1
					}
				}
				# foundry again third
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_ktisma_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_ktisma_foundry
							value > 0
						}
					}
					remove_district = district_giga_birch_ktisma_foundry
					set_or_change_variable = {
						which = giga_birch_district_ktisma_built
						value = -1
					}
				}
				# trade last
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_ktisma_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_ktisma_trade
							value > 0
						}
					}
					remove_district = district_giga_birch_ktisma_trade
					set_or_change_variable = {
						which = giga_birch_district_ktisma_built
						value = -1
					}
				}
				# supertensiles
				if = {
					limit = {
						check_variable = {
							which = giga_birch_district_ktisma_built
							value > 10
						}
						num_districts = {
							type = district_giga_birch_ktisma_supertensiles
							value > 0
						}
					}
					remove_district = district_giga_birch_ktisma_supertensiles
					set_or_change_variable = {
						which = giga_birch_district_ktisma_built
						value = -1
					}
				}
			}

			# physma
			while = {
				count = value:giga_birch_restoration_physma_unity
				add_district = district_giga_birch_physma_administration
				set_or_change_variable = {
					which = giga_birch_district_physma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_physma_sanctuary
				add_district = district_giga_birch_physma_administration
				set_or_change_variable = {
					which = giga_birch_district_physma_built
					value = 1
				}
			}
			while = {
				count = value:giga_birch_restoration_physma_farming
				add_district = district_giga_birch_physma_farm

					which = giga_birch_district_physma_built
					value = 1
				}

			while = {
				count = value:giga_birch_restoration_physma_laboratory
				add_district = district_giga_birch_physma_laboratory

				}

				if = {

					set_or_change_variable = {
						which = giga_birch_district_physma_built
						value = -1
					}
				}

				# unity second
				if = {
					limit = {

							which = giga_birch_district_physma_built

						}
						num_districts = {
							type = district_giga_birch_physma_farm
							value > 0
						}
					}
					remove_district = district_giga_birch_physma_farm
					set_or_change_variable = {
						which = giga_birch_district_physma_built
						value = -1
					}
				}
			}

			############################################################################
			# replace features

			# home stuff
			add_deposit = d_giga_birch_origin_home_insula_restored
			remove_deposit = d_giga_birch_origin_home_insula
			remove_deposit = d_giga_birch_origin_home_farms
			remove_deposit = d_giga_birch_origin_home_catalytic_farms
			remove_deposit = d_giga_birch_origin_home_mines
			remove_deposit = d_giga_birch_origin_home_lithoid_mines
			remove_deposit = d_giga_birch_origin_home_generators
			remove_deposit = d_giga_birch_origin_home_machine_generators
			remove_deposit = d_giga_birch_origin_home_labs

			# expedition 1
			add_deposit = d_giga_birch_expedition_1_restored
			remove_deposit = d_giga_birch_expedition_1_factory
			remove_deposit = d_giga_birch_expedition_1_farmers
			remove_deposit = d_giga_birch_expedition_1_miners
			remove_deposit = d_giga_birch_expedition_1_technicians

			# expedition 2
			add_deposit = d_giga_birch_expedition_2_restored
			remove_deposit = d_giga_birch_expedition_2_exterminate
			remove_deposit = d_giga_birch_expedition_2_trade
			remove_deposit = d_giga_birch_expedition_2_integrate
			remove_deposit = d_giga_birch_expedition_2_conquer
			remove_deposit = d_giga_birch_expedition_2_sanctuary

			remove_deposit = d_giga_birch_expedition_3

			remove_deposit = d_giga_birch_expedition_5

			# expedition 8
			add_deposit = d_giga_birch_expedition_8_restored
			remove_deposit = d_giga_birch_expedition_8

			# expedition 9
			add_deposit = d_giga_birch_expedition_9_restored
			remove_deposit = d_giga_birch_expedition_9_energy

			############################################################################
			# notification
			carrier_event = {
				id = giga_birch.10001
			}
			add_modifier = {
				modifier = repaired_birch_build_speed_buff
				days = 1800
			}
		}
	}
