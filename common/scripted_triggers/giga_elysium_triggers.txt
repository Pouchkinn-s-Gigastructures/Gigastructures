# elysium triggers

giga_planet_can_support_elysium = {
    custom_tooltip = { fail_text = "requires_not_star" is_star = no }
    custom_tooltip = { fail_text = "requires_not_astral_scar" is_astral_scar = no }
    custom_tooltip = { fail_text = "requires_not_a_megastructure" not = { has_carrier_flag = megastructure } }

    giga_elysium_has_supported_main_district = yes
}

giga_planet_has_elysium = {
    any_moon = {
        is_planet_class = pc_giga_elysium
    }
}

giga_planet_is_elysium_host = {
    has_modifier = giga_elysium_host_planet
}

# main district check
# does the target planet have any supported main district?
# tragically we need to check this as the generic way of doing it is hampered by edge cases, since the first district
# in a planet's list is not always the main one
giga_elysium_has_supported_main_district = {
    #custom_tooltip_fail = {

        # the or is important, script contains multiple triggers
        or = {
            inline_script = {
                script = megastructures/elysium/giga_elysium_supported_districts
            }
        }
    #}
}
