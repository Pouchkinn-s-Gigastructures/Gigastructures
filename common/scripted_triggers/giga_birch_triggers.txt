giga_birch_district_allow = {
	custom_tooltip_fail = {
		text = giga_birch_district_$category$_allow
		check_variable = {
			which = giga_birch_district_$category$_built
			value < modifier:giga_birch_district_$category$_max
		}
	}
}

# insula presence triggers

has_giga_birch_insula = {
	or = {
		has_district = district_giga_birch_$TYPE$
		has_district = district_giga_birch_$TYPE$_1
	}
}
