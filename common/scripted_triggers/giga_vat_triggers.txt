giga_is_vat_ship = { #All vat grown ships
    OR = {
        is_ship_size = giga_space_dragon
        is_ship_size = giga_wraith
        is_ship_size = giga_elder_tiyanki
        is_ship_size = giga_stellarite
        is_ship_size = giga_voidspawn
        is_ship_size = giga_queen
        is_ship_size = giga_large_ship
        is_ship_size = giga_large_ship_carrier
        is_ship_size = giga_small_ship
        is_ship_size = giga_toxic_god
        is_ship_size = giga_space_whale
        is_ship_size = giga_space_amoeba_centenarian
    }
}
