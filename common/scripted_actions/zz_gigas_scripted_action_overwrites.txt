horizon_needle_action = {
	user_scope = fleet
	scope = planet

	possible = {
		#hidden_trigger = {
		custom_tooltip = {
			fail_text = cosmogenesis_exodus_tt
			OR = {
				AND = {
					is_planet_class = pc_core_quasar
					from.owner = {
						has_country_flag = exodus_ready
					}
				}
				AND = {
					is_planet_class = pc_core_black_hole
					from.owner = {
						has_country_flag = exodus_ready
					}
				}
				AND = {
					is_planet_class = pc_black_hole
					from.owner = {
						has_country_flag = exodus_ready
					}
				}
				AND = {
					is_colony = yes
					has_carrier_flag = cosmogenesis_pickup_planet
					from.owner = {
						NOT = { has_country_flag = exodus_ready }
					}
				}
			}
		}
	}

	button_clickable = {
		custom_tooltip = {
			fail_text = cosmogenesis_exodus_tt
			from.owner = {
				NOR = {
					has_country_flag = cosmogenesis_supermassive
					has_country_flag = cosmogenesis_vester

				}
			}
		}
	}

	button_visible = {
		owner = {
			OR = {
				is_nomadic = no
				has_country_flag = exodus_ready
			}
		}
	}

	on_completed = on_cosmogenesis_exodus_planet_reached
	tooltip = exodus_ship_button
	icon = GFX_fleet_action_button_active_reconnaissance
	icon_selected = GFX_fleet_action_button_active_reconnaissance_selected
	activity_key = exodus_action_activity_key
	context_menu_name = exodus_context_menu_title
	progress_activity_key = exodus_action_progress_activity_key
	required_progress = 60
}

		}
	}
