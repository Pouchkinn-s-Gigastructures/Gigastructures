giga_ehof_resource_module = {
    icon = "GFX_waystation_ehof_resource_module"
    section = "REFINERY_STARBASE_SECTION"
    construction_days = @waystation_build_time_180
    starbase_type = waystation
    category = sm_economy_cat

    starbase_limit = {
        base = 1
    }

    potential = {
        is_waystation_starbase = yes
        owner? = {
            OR = {
                has_technology = tech_negative_e_s
                has_technology = tech_ehof_sentient_tier_1
            }
        }
    }
    show_in_tech = tech_negative_e_s
    show_tech_unlock_if = {
        AND = {
            NOT = { has_global_flag = ehof_disabled }
            is_nomadic = yes
        }
    }

    possible = {
        #sentient metal not yet discovered -> failtext only shows negative mass
        if = {
            limit = {
                owner? = {
                    NOT = { has_technology = tech_ehof_sentient_tier_1 }
                }
            }
            custom_tooltip =  {
                fail_text = requires_negative_mass_deposit
                AND = {
                    owner? = {
                        AND = {
                            has_technology = tech_negative_e_s
                            NOT = { has_technology = tech_ehof_sentient_tier_1 }
                        }
                    }
                    system_has_station_resource = {
                        RESOURCE = giga_sr_negative_mass
                    }
                }
            }
        }
        else = {
            #both resources discovered -> failtext shows both
            custom_tooltip = {
                fail_text = requires_negative_mass_or_sentient_metal_deposit
                AND = {
                    owner? = {
                        AND = {
                            has_technology = tech_ehof_sentient_tier_1
                            has_technology = tech_negative_e_s
                        }
                    }
                    OR = {
                        system_has_station_resource = {
                            RESOURCE = giga_sr_negative_mass
                        }
                        system_has_station_resource = {
                            RESOURCE = giga_sr_sentient_metal
                        }
                    }
                }
            }
        }
    }

    #Resources
    ##same cost & upkeep as refinery module.
    resources = {
        category = starbase_waystations_modules
        cost = {
            alloys = @waystation_cost_alloys_100
        }
        upkeep = {
            trade = @waystation_upkeep_trade_2
        }
    }

    collected_system_resources = {
        giga_sr_negative_mass = {
            owner? = {
                has_technology = tech_negative_e_s
            }
        }
        giga_sr_sentient_metal = {
            owner? = {
                has_technology = tech_ehof_sentient_tier_1
            }
        }
    }

    #Custom Tooltips
    triggered_station_modifier = {
        potential = {
            owner? = {
                AND = {
                    has_technology = tech_negative_e_s
                    NOT = { has_technology = tech_ehof_sentient_tier_1 }
                }
            }
        }
        custom_tooltip = allow_negative_mass_collection
        show_only_custom_tooltip = no
    }
    triggered_station_modifier = {
        potential = {
            owner? = {
                AND = {
                    has_technology = tech_negative_e_s
                    has_technology = tech_ehof_sentient_tier_1
                }
            }
        }
        custom_tooltip = allow_negative_mass_or_sentient_metal_collection
        show_only_custom_tooltip = no
    }

    #Stockpile modifiers
    triggered_station_modifier = {
        potential = {
            owner? = {
                has_technology = tech_negative_e_s
            }
            system_has_station_resource = {
                RESOURCE = giga_sr_negative_mass
            }
        }
        starbase_stockpile_giga_sr_negative_mass_capacity_add = @waystation_stockpile_125
    }
    triggered_station_modifier = {
        potential = {
            owner? = {
                has_technology = tech_ehof_sentient_tier_1
            }
            system_has_station_resource = {
                RESOURCE = giga_sr_sentient_metal
            }
        }
        starbase_stockpile_giga_sr_sentient_metal_capacity_add = @waystation_stockpile_125
    }

    #Wayline pact
    triggered_system_modifier = {
        potential = {
            exists = owner
            owner? = {
                NOT = {
                    has_technology = tech_ehof_sentient_tier_1
                }
            }
            space_owner? = {
                NOT = {
                    benefits_from_waystation_network = {
                        NOMAD = root.owner
                        SYSTEM_OWNER = this
                    }
                }
            }
        }
        #only tooltip needs to change
        custom_tooltip = waystation_collection_penalty_negative_mass_tt
        show_only_custom_tooltip = yes
        station_gatherers_giga_sr_negative_mass_produces_mult = -0.25
        station_gatherers_giga_sr_sentient_metal_produces_mult = -0.25
    }
    triggered_system_modifier = {
        potential = {
            exists = owner
            owner? = {
                has_technology = tech_ehof_sentient_tier_1
            }
            space_owner? = {
                NOT = {
                    benefits_from_waystation_network = {

                    }
                }
            }
        }

        station_gatherers_giga_sr_negative_mass_produces_mult = -0.25

    }

            }
            has_neighboring_waystation = yes
            system_has_station_resource = {

            }
        }

        show_only_custom_tooltip = no
        starbase_collected_system_giga_sr_sentient_metal_add = @waystation_joyous_path_add_4
    }
    triggered_station_modifier = {
