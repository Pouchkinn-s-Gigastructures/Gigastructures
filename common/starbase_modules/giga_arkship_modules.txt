        owner? = {
            has_technology = tech_abstract_1
        }
        is_arkship_starbase = yes
    }

    show_in_tech = "tech_arkship_exodus_jump"

    on_finished = {
        owner = {
            set_country_flag = ehof_spinal_mount_possessed
            set_country_flag = ehof_restored

            set_variable = {
                which = ehof_phase
                value = 1
            }
        }
        random_country = {
            limit = { is_country_type = global_event }
            change_variable = { which = amount_of_ehofs value = 1 }
        }
    }

    on_destroyed = {
        owner = {
            remove_country_flag = ehof_spinal_mount_possessed
        }

            limit = { is_country_type = global_event }
            change_variable = { which = amount_of_ehofs value = -1 }
        }
    }

        is_nomadic = yes
