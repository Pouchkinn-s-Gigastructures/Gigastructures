redtoxin = {
	trigger = {
		owner = {
			has_technology = tech_scourge_deployment
			ehof_default_country = yes
			NOT = { has_policy_flag = orbital_bombardment_selective }
		}
	}
	default = no

	stop_when_armies_dead = no
	abduct_pops = no
	planet_damage = 1.0
	army_damage = 1.5

	kill_pop_chance = {
		base = 1.5 # Same as pox, not limited to organics
	}
	kill_pop_amount = { min = 0 max = 200 }
	min_pops_to_kill_pop = 0

	# root = fleet
	# from = planet
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0.01
			exists = from
			from = {
				owner = { NOT = { is_hostile = root.owner } }
				NAND = {
					controller = { is_crisis_faction = yes }
					planet_garrison_strength > 0
				}
			}
		}
	}
}

			has_technology = tech_katzen_atomic_munitions

		}
	}

	stop_when_ground_combat = no

		base = 5.0 # Yeah this seems like a bombardment stance that will kill
	}
	min_pops_to_kill_pop = 0
	kill_pop_amount = { min = 0 max = 200 }

	# root = fleet
	# from = planet
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0.01
			exists = from
			from = {
				owner = { NOT = { is_hostile = root.owner } }
				NAND = {
					controller = { is_crisis_faction = yes }
					planet_garrison_strength > 0
				}
			}
		}
		modifier = {
			factor = 5 #Scorched Earth autochthony path favours spreading the flames
			owner = {
				has_country_flag = giga_scorched_habitability_preference #stealing this flag because it's set in the right way

		}
		modifier = {
			factor = 10000
			owner = {
