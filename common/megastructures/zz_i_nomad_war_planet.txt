nomad_war_planet_0 = {
	entity = "construction_platform_entity"
	construction_entity = "construction_platform_entity"
	# construction_blocks_and_blocked_by = none # BLOCKING TEST
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	prerequisites = { giga_tech_nomad_war_planet }
    build_type = inside_gravity_well

	outliner_trigger = {
		exists = owner
		NOT = { has_megastructure_flag = giga_outliner_hidden_by_@owner }
	}
	
	build_time = 1800
	resources = {
		category = giga_gigastructures
		cost = {
			influence = 150
			unity = @giga_giga_unity_cost
		}
		inline_script = {
			script = megastructures/generic_parts/giga_mega_alloy_cost
			alloys = 10000
		}
		upkeep = { energy = 5 }
	}

	potential = {
		has_technology = giga_tech_nomad_war_planet
		NOT = { has_global_flag = warplanet_disabled }
		OR = {
			has_global_flag = warplanet_capped_u
			check_variable = {
				which = giga_current_warplanet
				value < value:giga_warplanet_limit
			}
		}
	}

    possible = {
        fromfrom = {
            is_arkship_fleet = yes
        }
		custom_tooltip = {
			fail_text = "requires_not_capped"
			from = {
				OR = {
					has_global_flag = warplanet_capped_u
					check_variable = {
						which = giga_current_warplanet
						value < value:giga_warplanet_limit
					}
				}
			}
		}
    }

	ai_weight = {
		weight = value:giga_ai_base_giga

		mult = value:giga_ai_upkeep|RESOURCE|energy|AMOUNT|500|MIN|0.2|
		mult = value:giga_ai_upkeep|RESOURCE|alloys|AMOUNT|250|MIN|0.1|
		mult = value:giga_ai_core_territory
		mult = value:giga_ai_prevent_spam|MEGA|war_planet|
		mult = value:giga_ai_militarist_plus

		modifier = { factor = value:giga_ai_base_continue	from = { has_country_flag = pangalactic_defense_league } }
		modifier = { factor = 2		from = { has_federation = yes federation = { has_federation_type = military_federation } } }
		modifier = { factor = 1.5	from = { is_homicidal = yes } }
		modifier = { factor = 2		from = { is_at_war = yes } }
	}

	on_build_complete = {
		save_event_target_as = giga_system
		fromfrom= { save_event_target_as = giga_mega_targ }
		from = {
			set_timed_country_flag = { flag = has_recently_built_war_planet years = 10 }
			country_event = { id = giga_nomad_dialog.010 }	# Notification
			change_variable = { which = giga_current_warplanet value = 1 }
		}
	}
}

                        name = random
                        graphical_culture = root.from
                        random_existing_design = giga_nomad_war_planet
                        effect = {
                            set_ship_flag = giga_planet
                        }
                    }
                    set_formation_scale = 10
                    set_name = last_created_ship
                }
            }
        }
		last_created_fleet = { set_location = fromfrom }
		remove_megastructure = fromfrom
	}
}

    entity = "giga_nomad_war_planet_destroyed_entity"
    portrait = "GFX_megastructure_dyson_gun_background"
    place_entity_on_planet_plane = no
    potential = { always = no }
    resources = { category = giga_gigastructures }

    outliner_trigger = {
        exists = owner
        NOT = { has_megastructure_flag = giga_outliner_hidden_by_@owner }
    }

    dismantle_potential = { always = yes }

    dismantle_possible = {
        can_dismantle_megastructure = {
            TECH = tech_mega_engineering
        }
    }

    dismantle_cost = {
        category = giga_gigastructures
        cost = {
            energy = 30000
        }
    }

    dismantle_time = 3000

    on_dismantle_complete = {
        from = {
            add_resource = {
                alloys = 100000
                mult = value:giga_megastructure_dismantle_refund_mult # Vanilla megastructure_dismantle_refund_mult not used as all empires should get a reward from dismantling ruined warships, scavengers just get more
            }
            country_event = { id = giga_nomad_actions.552 }	# Notification
        }
    }
}
