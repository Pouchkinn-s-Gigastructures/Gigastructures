# Lunar Speculorefractor (disco moon) functions
# each mode swaps the modifier on the planet the disco moon orbits and that planet's other moons
# the AI keeps the lightshow, the highest weight, which is also the starting mode

giga_overclock_disco_lightshow = {
	icon = GFX_overclock_gc_default
	custom_tooltip = giga_overclock_disco_lightshow_effect
	on_overclock_start = {
		giga_disco_moon_set_mode = { MODIFIER = disco_moon_modifier }
	}
	resources = {
		category = giga_megastructures
		upkeep = {
			rare_crystals = 5
			energy = 50
			alloys = 25
		}
	}
	ai_weight = { weight = 100 }
}
