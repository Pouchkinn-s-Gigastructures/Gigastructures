# these overclocks are for selecting the mode for branching megas
giga_overclock_fusion_suppressor_supernova = {
	icon = GFX_overclock_gc_default
	custom_tooltip = giga_overclock_fusion_suppressor_supernova_effect
	on_overclock_start = {
		remove_megastructure_flag = black_hole
	}
	on_overclock_end = {
		set_megastructure_flag = black_hole
	}
	resources = {
		category = megastructures
		cost = {
			energy = @arc_furnace_overclock_switch_cost
		}
	}
}
