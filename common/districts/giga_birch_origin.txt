district_giga_origin_birch_lab = {
	base_buildtime = 360
	icon = district_giga_archaeology
	gridbox = district_arcology_research
	district_background = district_arcology_research
	is_uncapped = {
		always = no
	}

	zone_slots = {
		slot_birch_archaeology
	}

	show_on_uncolonized = {
		uses_district_set = standard
		is_planet_class = pc_origin_birch
	}

	potential = {
		uses_district_set = standard
		is_planet_class = pc_origin_birch
	}

	prerequisites = {
		tech_basic_science_lab_1
	}

	show_tech_unlock_if = {
		has_origin = origin_birch
	}

	resources = {
		category = planet_districts

		cost = {
			minerals = 400
		}

		upkeep = {
			energy = 3
		}

		# expedition 2
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_2_complete
				or = {
					has_carrier_flag = birch_expedition_2_exterminate
					has_carrier_flag = birch_expedition_2_trade
				}
			}
			unity = 5
			society_research = 8
			multiplier = value:giga_job_scaling_plus_base
		}

		# expedition 3
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_3_physics
			}
			physics_research = 6
			multiplier = value:giga_job_scaling_plus_base
		}
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_3_engineering
			}
			engineering_research = 6
			multiplier = value:giga_job_scaling_plus_base
		}
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_3_artifacts
			}
			minor_artifacts = 0.2
			multiplier = value:giga_job_scaling_plus_base
		}

		# expedition 7
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_7_science
				has_carrier_flag = birch_expedition_7_complete
			}
			physics_research = 3
			society_research = 3
			engineering_research = 3
			multiplier = value:giga_job_scaling_plus_base
		}
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_7_unity
				has_carrier_flag = birch_expedition_7_complete
			}
			unity = 5
			multiplier = value:giga_job_scaling_plus_base
		}

		# expedition 7 trade
		produces = {
			trigger = {
				has_carrier_flag = birch_expedition_7_trade
				has_carrier_flag = birch_expedition_7_complete
				exists = owner
			}
			trade = 15
			mult = value:giga_job_scaling_plus_base
		}

		# visuals handled in the zone
		# why cant the zone actually produce?? idk
		produces = {
			trigger = {
				has_zone = { zone = zone_giga_origin_birch_lab }
			}
			minor_artifacts = 0.33
		}
	}

	triggered_desc = {
		trigger = {
			exists = owner
			owner = { has_origin = origin_birch }
		}
		text = district_giga_origin_birch_lab_tooltip
	}

	# 150 total researchers
	inline_script = {
		script = jobs/researchers_add
		AMOUNT = 50
	}

	# 150 unity jobs
	# why is there no inline for this thats not zones bruh
	# copied this from that big inline
	### Regular Empires
	# Planets
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_regular_empire = yes
			}
		}
		job_bureaucrat_add = 150
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_active_tradition = tr_virtuality_4
				is_regular_empire = yes
			}
		}
		job_bureaucrat_add = 150
		mult = 0.5
	}

	# Planets
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = yes
				NOT = {
					has_valid_civic = civic_machine_servitor
				}
			}
		}
		job_coordinator_add = 150
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_active_tradition = tr_virtuality_4
				is_gestalt = yes
				NOT = {
					has_valid_civic = civic_machine_servitor
				}
			}
		}
		job_coordinator_add = 150
		mult = 0.5
	}

	# Planets
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_valid_civic = civic_machine_servitor
			}
		}
		job_bio_trophy_add = 150
		mult = 5
	}

}
