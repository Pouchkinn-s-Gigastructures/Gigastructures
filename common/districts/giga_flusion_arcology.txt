#Housing: Amenities
district_flusion_city = {
	base_buildtime = 480
	is_capped_by_modifier = no
	exempt_from_ai_planet_specialization = yes
    overlay_icon = GFX_district_government
    gridbox = district_city
    district_background = district_city
        # i think this looks better than the arcology housing one

	show_on_uncolonized = {
		is_planet_class = pc_flusion_gaia_ecu
		uses_district_set = flusion_gaia_ecu
	}

	potential = {
		exists = owner
		is_planet_class = pc_flusion_gaia_ecu
		uses_district_set = flusion_gaia_ecu
	}

	allow = {
		NOR = {
			has_modifier = resort_colony
			has_modifier = slave_colony
		}
	}

	conversion_ratio = 1
	convert_to = {
		district_city_katzen
		district_city
	}

	zone_slots = {
		slot_city_government
		slot_city_01
		slot_city_02
	}

	resources = {
		category = planet_districts_industrial
		cost = {
			minerals = 720
		}
		upkeep = {
			energy = 2
		}
	}

	planet_modifier = {
		planet_housing_add = 2400
		planet_amenities_add = 2800
		planet_orbital_bombardment_damage = -0.05
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				OR = {
					is_country_type = katzenartig_imperium
					is_gestalt = no
				}
			}
		}
		modifier = {
			job_enforcer_add = 400
			job_soldier_add = 100
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = yes
			}
		}
		modifier = {
			job_patrol_drone_add = 400
			job_warrior_drone_add = 100
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_active_tradition = tr_prosperity_public_works }
		}
		modifier = {
			planet_housing_add = 300
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_housing_1
			}
		}
		modifier = {
			planet_housing_add = 300
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_housing_2
				NOT = { has_valid_civic = civic_agrarian_idyll }
			}
		}
		modifier = {
			planet_housing_add = 300
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = no
				has_active_tradition = tr_mercantile_trickle_up_economics
				is_tankbound_empire = no
			}
		}
		modifier = {
			job_clerk_add = 300
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = no
				has_active_tradition = tr_mercantile_trickle_up_economics
				is_tankbound_empire = yes
			}
		}
		modifier = {
			job_production_overseer_add = 200
		}
	}
	
	# Bug Branch
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { 
				is_gestalt = no
				is_bugged_rooftop_farmers = yes
			}
		}
		modifier = {
			# the roofs are full of AA guns already
			job_enforcer_add = -100
			job_farmer_add = 100
		}
	}

	# Bug Branch


	inline_script = {
		script = planet/research/giga_researcher_job_swap
		jobs = 600
		district = @yes
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				OR = {
					is_country_type = katzenartig_imperium
					is_gestalt = no
				}
			}
		}
		modifier = {
			job_chemist_add = 100
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_gestalt = yes
			}
		}
		modifier = {
			job_chemist_drone_add = 100
		}
	}
}
